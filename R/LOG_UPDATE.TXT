HOW TO USE catviz (quick reference)
================================================================================

## Step 1: Build spec
spec <- cat_spec(data, id="hospital_id", time="year", g="g",
                 subgroup="p",   # omit for pure CSDiD
                 group_id="state")

## Step 2: Label cohorts (optional but recommended)
spec <- cat_label(spec)

## Step 3: Count nodes
cat_counts(spec)

## Step 4: Plot (auto-selects DiD / CSDiD / DDD)
p <- cat_plot_tree(spec, counts=TRUE)
p <- cat_plot_tree(spec, grayscale=TRUE)   # B&W version

## Direct access to specific plots
cat_plot_did(spec)        # 2x2 DiD
cat_plot_csdid(spec)      # CSDiD
cat_plot_ddd(spec)        # DDD

## Save
cat_save_png(p, "tree.png", width=14, height=9, dpi=400)

## Diagnostics
cat_diag(spec, outcome="outcome_var", method="csdid")
cat_diag(spec, outcome="outcome_var", method="drddd")
cat_diag(spec, method="event")

## ATT equations
cat_att_equation("drddd")
cat_att_equation("csdid", subgroup_value=1)

## Balance
bal <- cat_balance_table(spec, covariates=c("var1","var2"), by="node")
cat_balance_plot(bal)
